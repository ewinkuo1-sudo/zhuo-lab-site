{{- /* Builds research-map/data.js for the static research map page. Map-specific text and figures come from data/research_map.json. Everything the map shares with the rest of the site is taken from the same files the Hugo pages use, so a new paper or a renamed station is edited once: statement, description <- the home page hero question, method, finding, status <- data/research.json papers <- data/research.json (slugs) + the publication pages verb, color <- data/stations.json cases <- data/cases.json nav, official <- the site menu and home URL Called once per build from hooks/head-end. */ -}} {{- $en := index (where site.Sites "Language.Lang" "en") 0 -}} {{- $out := dict -}} {{- range $lang := slice "zh" "en" -}} {{- $src := index site.Data.research_map $lang -}} {{- $langSite := index (where site.Sites "Language.Lang" $lang) 0 -}} {{- $stations := slice -}} {{- range $s := $src.stations -}} {{- $id := index $s "key" -}} {{- $papers := slice -}} {{- range $slug := (index site.Data.research $id).papers -}} {{- $pub := $en.GetPage (printf "/publication/%s" $slug) -}} {{- if not $pub -}}{{- errorf "research map: missing publication %s" $slug -}}{{- end -}} {{- $first := index $pub.Params.authors 0 -}} {{- $last := index (last 1 (split $first " ")) 0 -}} {{- $papers = $papers | append (dict "title" $pub.Title "authors" (printf "%s%s" $last (cond (eq $lang "zh") " 等" " et al.")) "journal" ($pub.Params.publication | strings.TrimPrefix "*" | strings.TrimSuffix "*") "year" $pub.Date.Year "doi" $pub.Params.doi) -}} {{- end -}} {{- $st := index site.Data.stations $id -}} {{- $cases := slice -}} {{- range $c := site.Data.cases -}} {{- with index $c.stations $id -}} {{- $cases = $cases | append (dict "id" $c.id "name" (index $c $lang) "state" .) -}} {{- end -}} {{- end -}} {{- $text := index (index site.Data.research $id) $lang -}} {{- $stations = $stations | append (merge $s (dict "question" $text.question "method" $text.method "finding" $text.finding "status" $text.status "papers_label" $text.papers_label "short" (index $st $lang).short "papers" $papers "verb" (index $st $lang).verb "color" $st.color "cases" $cases)) -}} {{- end -}} {{- $topCases := slice -}} {{- range $c := site.Data.cases -}} {{- $topCases = $topCases | append (dict "id" $c.id "name" (index $c $lang) "stations" $c.stations) -}} {{- end -}} {{- $nav := slice -}} {{- range $m := $langSite.Menus.main -}} {{- if not $m.HasChildren -}} {{- with $langSite.GetPage $m.URL -}}{{- $nav = $nav | append (dict "name" $m.Name "url" .Permalink) -}}{{- end -}} {{- end -}} {{- end -}} {{- $hero := (index $langSite.Home.Params.sections 0).content -}} {{- $out = merge $out (dict $lang (merge $src (dict "lang" $lang "statement" $hero.statement "description" $hero.description "stations" $stations "cases" $topCases "official" $langSite.Home.Permalink "nav" $nav))) -}} {{- end -}} {{- $js := printf "// Generated by Hugo from data/research_map.json and the site data. Do not edit.\nwindow.SITE_DATA=%s;\n" ($out | jsonify) -}} {{- (resources.FromString "research-map/data.js" $js).Publish -}}
